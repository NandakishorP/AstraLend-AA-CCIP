\section{Background}

\begin{frame}{Background: Blockchain Fundamentals}
    \framesubtitle{A distributed, append-only ledger with no controlling authority}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{itemize}
            \item \textbf{Shared} --- thousands of nodes, identical copies
            \item \textbf{Append-only} --- no edits, no deletions
            \item \textbf{No owner} --- no head office, no reversals
        \end{itemize}
        \vspace{0.4cm}
        \begin{itemize}
            \item Bank ledger $\to$ trusted \emph{because of} the bank
            \item Blockchain $\to$ trusted \emph{without} one
        \end{itemize}

\begin{frame}{Background: Principles of Secured Lending}
    \framesubtitle{Established banking practices replicated by the protocol}
    \vspace{0.15cm}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{1. Deposits fund loans}
        \begin{itemize}
            \item Savers pool money
            \item Borrowers draw from it
        \end{itemize}
        \vspace{0.3cm}
        \textbf{2. Collateral secures}
        \begin{itemize}
            \item Gold loan, loan against FD
            \item Default $\to$ lender recovers
        \end{itemize}
        \vspace{0.3cm}
        \textbf{3. Haircut on value}
        \begin{itemize}
            \item Pledge \$100 $\to$ get \$75
            \item The gap = \textbf{loan-to-value}
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{4. Act before cushion is gone}
        \begin{itemize}
            \item Value falls $\to$ margin call
            \item Then sale = \textbf{liquidation}
        \end{itemize}
        \vspace{0.3cm}
        \textbf{Rates follow demand}
        \begin{itemize}
            \item Most deposits lent $\to$ rates up
            \item Money idle $\to$ rates down
            \item = \textbf{utilisation}
        \end{itemize}
        \vspace{0.35cm}
        \keypoint{These four parameters govern the protocol design.}
    \end{columns}
\end{frame}

\section{Problem Statement}

\begin{frame}{Problem Statement: Liquidity Fragmentation}
    \begin{columns}
        \column{0.5\textwidth}
        \begin{itemize}
            \item Hundreds of blockchains
            \item Each keeps its own ledger
            \item \textbf{None can read another's}
        \end{itemize}
        \vspace{0.35cm}
        \textbf{Banking parallel}
        \begin{itemize}
            \item FD at Bank A
            \item Loan wanted from Bank B
            \item B cannot see or verify it
        \end{itemize}
        \vspace{0.35cm}
        \begin{itemize}
            \item Banks have credit bureaus, CERSAI
            \item \textbf{Blockchains have no shared layer}
        \end{itemize}

        \column{0.5\textwidth}
        \centering
        \resizebox{0.92\linewidth}{!}{%
        \begin{tikzpicture}[font=\sffamily\small,
            chain/.style={draw=slate, fill=white, rounded corners=3pt, text width=3.1cm, align=center, inner sep=7pt, minimum height=1.8cm},

        ]

        \end{tikzpicture}}
    \end{columns}
    \vspace{0.2cm}
    \keypoint{Capital remains idle on one network while demand is unmet on another.}
\end{frame}

\begin{frame}{Existing Approach: Asset Bridging}
    \framesubtitle{Mechanism and its structural limitations}
    \begin{columns}
        \column{0.5\textwidth}
        \textbf{How a bridge works}
        \begin{enumerate}
            \item Lock asset on Chain A
            \item Mint substitute on Chain B
            \item Use substitute as collateral
        \end{enumerate}
        \vspace{0.35cm}
        \textbf{Why it is fragile}
        \begin{itemize}
            \item \textbf{Two objects, one value}
            \item Correctness rests on the bridge's books
            \item Mint without lock $\to$ counterfeit
        \end{itemize}
        \vspace{0.25cm}
        \begin{itemize}
            \item Largest loss category in the sector
            \item \textbf{Billions of dollars}
        \end{itemize}

        \column{0.5\textwidth}
        \centering
        \resizebox{0.95\linewidth}{!}{%
        \begin{tikzpicture}[font=\sffamily\small,

\section{Proposed Solution}

\begin{frame}{Proposed Approach}
    \vspace{0.5cm}
    \centering
    \colorbox{amritaMaroon}{\parbox{0.92\linewidth}{\vspace{0.25cm}\centering
      \color{white}\large \textbf{Do not move the money.\\[4pt] Move the record of it.}\vspace{0.25cm}}}
    \vspace{0.6cm}

    \begin{columns}
        \column{0.48\textwidth}
        \textbf{Banking parallel}
        \begin{itemize}
            \item Bank A does not courier the gold
            \item It sends a \textbf{confirmation}
            \item Bank B lends against that
        \end{itemize}

        \column{0.48\textwidth}
        \textbf{What this buys}
        \begin{itemize}
            \item No substitute token anywhere
            \item Nothing to reconcile
            \item Bridge failures \textbf{structurally absent}
        \end{itemize}
        \vspace{0.2cm}
        \textbf{Trade-off:} remote state is consistent shortly after the event, not instantly
    \end{columns}
\end{frame}

% --- INTEREST & RISK ---
\begin{frame}{Interest Accrual, Risk Parameters and Liquidation}
    \framesubtitle{Protocol risk engine}
    \begin{columns}
        \column{0.5\textwidth}
        \small
        \textbf{Interest --- one number per asset}
        \begin{itemize}
            \item \code{index += index $\times$ rate $\times$ dt}
            \item Loan stores index at origination
            \item \code{debt = principal $\times$ now / then}
            \item \textbf{Whole book accrues, O(1)}
            \item Interest cleared before principal
        \end{itemize}
        \vspace{0.25cm}
        \textbf{Rate curve}
        \begin{itemize}
            \item Base \textbf{5\%}, kink \textbf{70\%}, cap \textbf{100\%}
            \item Steep above kink $\to$ defends liquidity
        \end{itemize}

        \column{0.5\textwidth}
        \small
        \textbf{Liquidation --- automated}
        \begin{itemize}
            \item Monitors collateral value + due date
            \item Chainlink Automation triggers it
        \end{itemize}
        \vspace{0.25cm}
        \textbf{Escalates like a bank}
        \begin{itemize}
            \item Penalty 1 $\to$ +30 days
            \item Penalty 2 $\to$ +30 days
            \item Then seizure
            \item \textbf{Not liquidated on first breach}
        \end{itemize}
    \end{columns}
    \vspace{0.3cm}
    \keypoint{75\% LTV $\cdot$ 80\% liquidation threshold $\cdot$ 180-day term $\cdot$ live price feeds}
\end{frame}

            \item \textbf{SQLite} --- single writer, embedded, zero-ops
        \end{itemize}
        \vspace{0.25cm}
        \textbf{Patterns}
        \begin{itemize}
            \item Facade + delegation
            \item Proxy (EIP-1967) --- upgrade, keep storage
            \item Custody segregated from accounting
        \end{itemize}

        \column{0.5\textwidth}
        \textbf{Security}
        \begin{itemize}
            \item Sender allow-listing on every message
            \item Readers $\neq$ writers on mirrors
            \item Reentrancy guards, checks-effects-interactions
            \item Custom errors --- cheap, decodable
        \end{itemize}
        \vspace{0.25cm}
        \textbf{Performance}
        \begin{itemize}
            \item O(1) interest, no per-loan loops
            \item Append-only enum --- in-flight messages safe
            \item Indexed reads decouple UI from chain
        \end{itemize}
    \end{columns}
\end{frame}
